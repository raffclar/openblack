include(CTest)
include(GoogleTest)

# The tests build into one gtest executable per area (A2(b), 2026-10): one link each instead of one per test file.
# ctest still lists every TEST on its own (gtest_discover_tests); each group executable also runs whole.
# A test file goes into the group of its area below: one openblack_setup_and_add_test line (his macro name) after
# the group's openblack_test_group().

# raffclar's macro names, as thin wrappers over our groups: a test file is added to the group that
# openblack_test_group() opened, and openblack_setup_test_group() builds the group from those files. Unlike his
# macros these build no executable of their own: one exe per area keeps the PCH and one link per group.
macro (openblack_test_group GROUP_NAME)
  set(OPENBLACK_TEST_GROUP ${GROUP_NAME})
endmacro ()

macro (OPENBLACK_SETUP_AND_ADD_TEST TEST_NAME TEST_SOURCE)
  list(APPEND OPENBLACK_TEST_SOURCES_${OPENBLACK_TEST_GROUP} ${TEST_SOURCE})
endmacro ()

# A test with JSON scenario files: they are copied next to the build of the test's directory
macro (OPENBLACK_SETUP_AND_ADD_JSON_TEST TEST_NAME TEST_SOURCE)
  openblack_setup_and_add_test(${TEST_NAME} ${TEST_SOURCE})
  get_filename_component(_openblack_test_dir ${TEST_SOURCE} DIRECTORY)
  list(APPEND OPENBLACK_TEST_SCENARIOS_${OPENBLACK_TEST_GROUP} ${_openblack_test_dir})
endmacro ()

# The data and settings every test executable shares
function (openblack_setup_test_group GROUP_NAME)
  # vcpkg's DLL copy (applocal) once, for test_core: every group links the same DLLs into the same bin/Release
  if (NOT GROUP_NAME STREQUAL "test_core")
    set(VCPKG_APPLOCAL_DEPS OFF)
  endif ()
  add_executable(${GROUP_NAME} ${OPENBLACK_TEST_SOURCES_${GROUP_NAME}})
  # its support/TestServices.cpp object and precompiled header come from the group's flag set, at the end of the file
  set_property(GLOBAL APPEND PROPERTY OPENBLACK_TEST_GROUPS ${GROUP_NAME})
  # the test sources include support/, the golden headers and third_party/ from test/, wherever they are
  target_include_directories(${GROUP_NAME} PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
  add_dependencies(${GROUP_NAME} generate_mock_game_data)
  target_link_libraries(${GROUP_NAME} PRIVATE GTest::gtest_main openblack_lib)
  target_compile_definitions(
    ${GROUP_NAME} PRIVATE TEST_BINARY_DIR="${CMAKE_CURRENT_BINARY_DIR}"
                          GLM_ENABLE_EXPERIMENTAL
  )
  set_property(TARGET ${GROUP_NAME} PROPERTY FOLDER "tests")
  set_target_properties(${GROUP_NAME} PROPERTIES UNITY_BUILD OFF)
  # PRE_TEST: listed when ctest runs, so the build does not run each exe after its link. gtest_discover_tests marks
  # GTEST_SKIP tests as skipped by itself
  gtest_discover_tests(
    ${GROUP_NAME} DISCOVERY_MODE PRE_TEST DISCOVERY_TIMEOUT 60
    WORKING_DIRECTORY ${CMAKE_CURRENT_BINARY_DIR}
  )
  foreach (TEST_DIR_NAME IN LISTS OPENBLACK_TEST_SCENARIOS_${GROUP_NAME})
    openblack_copy_test_scenarios(${GROUP_NAME} ${TEST_DIR_NAME})
  endforeach ()
endfunction ()

# A test with JSON scenario files: they are copied next to the build of the test's directory
function (openblack_copy_test_scenarios GROUP_NAME TEST_DIR_NAME)
  add_custom_target(
    ${GROUP_NAME}_${TEST_DIR_NAME}_scenarios
    COMMAND
      ${CMAKE_COMMAND} -E copy_directory
      ${CMAKE_CURRENT_SOURCE_DIR}/${TEST_DIR_NAME}/scenarios
      ${CMAKE_CURRENT_BINARY_DIR}/${TEST_DIR_NAME}/scenarios
  )
  set_property(TARGET ${GROUP_NAME}_${TEST_DIR_NAME}_scenarios PROPERTY FOLDER "tests")
  add_dependencies(${GROUP_NAME} ${GROUP_NAME}_${TEST_DIR_NAME}_scenarios)
  target_include_directories(${GROUP_NAME} PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/third_party)
endfunction ()

# support/TestServices.cpp and the precompiled header are built once for each set of flags the groups have, and every
# group of that set links the one object and reuses the one header. A group that links bgfx directly gets bgfx's
# definitions and /Zc:preprocessor, so there are two sets
get_property(OPENBLACK_TEST_GROUPS GLOBAL PROPERTY OPENBLACK_TEST_GROUPS)
foreach (FLAG_SET plain bgfx)
  set(SUPPORT openblack_test_support_${FLAG_SET})
  add_library(${SUPPORT} OBJECT ${CMAKE_CURRENT_SOURCE_DIR}/support/TestServices.cpp)
  target_include_directories(${SUPPORT} PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
  target_link_libraries(${SUPPORT} PUBLIC GTest::gtest_main openblack_lib)
  if (FLAG_SET STREQUAL "bgfx")
    target_link_libraries(${SUPPORT} PUBLIC bgfx::bgfx)
  endif ()
  target_compile_definitions(
    ${SUPPORT} PRIVATE TEST_BINARY_DIR="${CMAKE_CURRENT_BINARY_DIR}" GLM_ENABLE_EXPERIMENTAL
  )
  set_property(TARGET ${SUPPORT} PROPERTY FOLDER "tests")
  set_target_properties(${SUPPORT} PROPERTIES UNITY_BUILD OFF)
  if (OPENBLACK_USE_PCH AND NOT OPENBLACK_CLANG_TIDY_CHECKS)
    target_precompile_headers(${SUPPORT} PRIVATE ${OPENBLACK_PCH_HEADERS} <gtest/gtest.h>)
  endif ()
endforeach ()
foreach (GROUP_NAME ${OPENBLACK_TEST_GROUPS})
  get_target_property(GROUP_LIBRARIES ${GROUP_NAME} LINK_LIBRARIES)
  if ("bgfx::bgfx" IN_LIST GROUP_LIBRARIES)
    set(SUPPORT openblack_test_support_bgfx)
  else ()
    set(SUPPORT openblack_test_support_plain)
  endif ()
  target_link_libraries(${GROUP_NAME} PRIVATE ${SUPPORT})
  if (OPENBLACK_USE_PCH AND NOT OPENBLACK_CLANG_TIDY_CHECKS)
    target_precompile_headers(${GROUP_NAME} REUSE_FROM ${SUPPORT})
  endif ()
endforeach ()

# The mod loader library on its own: nothing of the game, each test with a mods folder of its own
# (vcpkg's DLL copy is test_core's, as for the groups)
set(VCPKG_APPLOCAL_DEPS OFF)
add_executable(test_mod_loader test_mod_loader.cpp)
target_link_libraries(test_mod_loader PRIVATE GTest::gtest_main ModLoader)
set_property(TARGET test_mod_loader PROPERTY FOLDER "tests")
gtest_discover_tests(
  test_mod_loader DISCOVERY_MODE PRE_TEST DISCOVERY_TIMEOUT 60
  WORKING_DIRECTORY ${CMAKE_CURRENT_BINARY_DIR}
)
